% ============================================================
\begin{frame}{Problem}
\begin{itemize}
    \setlength\itemsep{0.8em}
    \item Single-node multi-GPU LLM serving: replicas can fail (GPU reset, OOM, driver crash\ldots)
    \item Failure $\rightarrow$ capacity loss $\rightarrow$ interrupted streams, SLO violations, goodput drop
    \item Naive checkpoint: too frequent = high overhead; too rare = long recovery
    \item \textbf{Core question:} how to jointly optimize \textcolor{myblue}{routing}, \textcolor{myblue}{admission}, and \textcolor{myblue}{adaptive checkpointing} to maximize goodput while meeting SLOs under failures?
\end{itemize}
\end{frame}

% ============================================================
\begin{frame}{Our System}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\textbf{\textcircled{1} Benders-based Robust Scheduler}
\begin{itemize}
    \item Periodic decision epochs on system snapshot
    \item Master: admission + routing
    \item Subproblem: verify recovery per failure scenario
    \item Pooled-flow screening + logic-based cuts
\end{itemize}

\vspace{0.5em}
\textbf{\textcircled{2} Adaptive Checkpoint Policy}
\begin{itemize}
    \item Per-request, triggered on new stable KV blocks
    \item Publish when: $\Delta\text{replay} > \Delta\text{load} + \lambda \cdot \Delta\text{ckpt}$
    \item Early: skip (cheap to replay)
    \item Late: checkpoint (expensive to lose)
\end{itemize}
\end{column}

\begin{frame}{Summary}
\textbf{Key Findings}
\begin{enumerate}
    \item \textbf{Near-zero normal-case overhead}: goodput matches No-FT
    \item \textbf{Best under failures}: highest recovery rate + lowest SLO violations
    \item \textbf{More checkpointing $\neq$ better}: Fixed-High hurts both goodput and recovery
    \item \textbf{Adaptive checkpoint is the primary contributor}; routing matters more at scale
    \item \textbf{Solver overhead is small}: $\sim$15ms per epoch
\end{enumerate}

\vspace{0.5em}
